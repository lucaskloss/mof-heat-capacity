\newcommand{\moffivepanels}[2]{%
  \begin{tikzpicture}
    \begin{groupplot}[
      group style={group size=3 by 1,horizontal sep=0.5cm,
        xlabels at=edge bottom,ylabels at=edge left,yticklabels at=edge left},
      scale only axis,
      width=\dimexpr(\textwidth-2.6cm)/3\relax,height=0.25\textwidth,
      xmin=195,xmax=405,ymin=0,ymax=1.8,
      xtick={200,250,300,350,400},ytick={0,0.4,0.8,1.2,1.6},
      xlabel={$T$ [K]},
      ylabel={$C_p$ [J\,g$^{-1}$\,K$^{-1}$]},
      tick label style={font=\small},label style={font=\small},
      grid=major,grid style={black!10},
      legend style={font=\scriptsize,draw=none,fill=none,at={(0,1.03)},anchor=south west},
    ]
      \moffivepanel{#1}{#2}{50}{mofloadingfifty}
      \moffivepanel{#1}{#2}{100}{mofloadinghundred}
      \moffivepanel{#1}{#2}{150}{mofloadinghundredfifty}
    \end{groupplot}
  \end{tikzpicture}%
}
