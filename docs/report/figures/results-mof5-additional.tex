\newcommand{\moffiveadditional}[3]{%
  \begin{tikzpicture}
    \begin{axis}[
      scale only axis,
      width=\dimexpr\textwidth-1.9cm\relax,height=0.29\textwidth,
      xmin=195,xmax=405,ymin=0,ymax=2,
      xtick={200,250,300,350,400},ytick={0,0.4,0.8,1.2,1.6,2},
      xlabel={$T$ [K]},ylabel={$C_p$ [J\,g$^{-1}$\,K$^{-1}$]},
      tick label style={font=\small},label style={font=\small},
      grid=major,grid style={black!10},
      legend style={font=\scriptsize,draw=none,fill=none,at={(0,1.03)},anchor=south west},
    ]
      \moffiveadditionalpanel{#1}{#2}{#3}
    \end{axis}
  \end{tikzpicture}%
}

    table[x=temperature_K,y=upper,col sep=comma]{figures/data/mof5-additional-empty.csv};
